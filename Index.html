<script> 

let audioCtx = null; 

let noiseBufferCache = null; 

function initAudioContext() { 

  if (!audioCtx) { 

    const AudioContextClass = window.AudioContext || window.webkitAudioContext; 

    if (AudioContextClass) { 

      audioCtx = new AudioContextClass(); 

    } 

  } 

  if (audioCtx && audioCtx.state === 'suspended') { 

    audioCtx.resume(); 

  } 

} 

function getNoiseBuffer() { 

  if (noiseBufferCache) return noiseBufferCache; 

  if (!audioCtx) return null; 

  const bufferSize = audioCtx.sampleRate * 0.4; 

  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate); 

  const data = buffer.getChannelData(0); 

  for (let i = 0; i < bufferSize; i++) { 

    data[i] = Math.random() * 2 - 1; 

  } 

  noiseBufferCache = buffer; 

  return buffer; 

} 

function soundPlayCard() { 

  initAudioContext(); 

  if (!audioCtx) return; 

  const now = audioCtx.currentTime; 

  const noise = audioCtx.createBufferSource(); 

  noise.buffer = getNoiseBuffer(); 

  const noiseFilter = audioCtx.createBiquadFilter(); 

  noiseFilter.type = 'bandpass'; 

  noiseFilter.frequency.setValueAtTime(2600, now); 

  noiseFilter.Q.setValueAtTime(2.2, now); 

  const noiseGain = audioCtx.createGain(); 

  noiseGain.gain.setValueAtTime(0.8, now); 

  noiseGain.gain.exponentialRampToValueAtTime(0.001, now + 0.055); 

  noise.connect(noiseFilter); 

  noiseFilter.connect(noiseGain); 

  noiseGain.connect(audioCtx.destination); 

  noise.start(now); 

  noise.stop(now + 0.06); 

  const bodyOsc = audioCtx.createOscillator(); 

  const bodyGain = audioCtx.createGain(); 

  bodyOsc.type = 'sine'; 

  bodyOsc.frequency.setValueAtTime(140, now); 

  bodyOsc.frequency.exponentialRampToValueAtTime(45, now + 0.07); 

  bodyGain.gain.setValueAtTime(0.5, now); 

  bodyGain.gain.exponentialRampToValueAtTime(0.001, now + 0.07); 

  bodyOsc.connect(bodyGain); 

  bodyGain.connect(audioCtx.destination); 

  bodyOsc.start(now); 

  bodyOsc.stop(now + 0.07); 

} 

function soundDrawCard() { 

  initAudioContext(); 

  if (!audioCtx) return; 

  const now = audioCtx.currentTime; 

  const noise = audioCtx.createBufferSource(); 

  noise.buffer = getNoiseBuffer(); 

  const filter = audioCtx.createBiquadFilter(); 

  filter.type = 'bandpass'; 

  filter.frequency.setValueAtTime(1000, now); 

  filter.frequency.linearRampToValueAtTime(1800, now + 0.06); 

  filter.frequency.exponentialRampToValueAtTime(800, now + 0.12); 

  filter.Q.setValueAtTime(1.8, now); 

  const gain = audioCtx.createGain(); 

  gain.gain.setValueAtTime(0.01, now); 

  gain.gain.linearRampToValueAtTime(0.45, now + 0.035); 

  gain.gain.exponentialRampToValueAtTime(0.001, now + 0.13); 

  noise.connect(filter); 

  filter.connect(gain); 

  gain.connect(audioCtx.destination); 

  noise.start(now); 

  noise.stop(now + 0.14); 

} 

function soundTurnClose() { 

  initAudioContext(); 

  if (!audioCtx) return; 

  const now = audioCtx.currentTime; 

  const noise = audioCtx.createBufferSource(); 

  noise.buffer = getNoiseBuffer(); 

  const filter = audioCtx.createBiquadFilter(); 

  filter.type = 'lowpass'; 

  filter.frequency.setValueAtTime(1400, now); 

  filter.frequency.exponentialRampToValueAtTime(350, now + 0.18); 

  const gain = audioCtx.createGain(); 

  gain.gain.setValueAtTime(0.3, now); 

  gain.gain.exponentialRampToValueAtTime(0.001, now + 0.18); 

  noise.connect(filter); 

  filter.connect(gain); 

  gain.connect(audioCtx.destination); 

  noise.start(now); 

  noise.stop(now + 0.19); 

} 

function soundShuffleCards() { 

  initAudioContext(); 

  if (!audioCtx) return; 

  let delay = 0; 

  for (let i = 0; i < 7; i++) { 

    setTimeout(() => { soundDrawCard(); }, delay * 1000); 

    delay += 0.038 + Math.random() * 0.02; 

  } 

} 

function soundTeasingPenalty() { 

  initAudioContext(); 

  if (!audioCtx) return; 

  const now = audioCtx.currentTime; 

  const notes = [587.33, 523.25, 440, 349.23]; 

  notes.forEach((freq, idx) => { 

    const osc = audioCtx.createOscillator(); 

    const gain = audioCtx.createGain(); 

    osc.type = 'triangle'; 

    osc.frequency.setValueAtTime(freq, now + idx * 0.1); 

    osc.frequency.exponentialRampToValueAtTime(freq * 0.85, now + idx * 0.1 + 0.09); 

    gain.gain.setValueAtTime(0.25, now + idx * 0.1); 

    gain.gain.exponentialRampToValueAtTime(0.01, now + idx * 0.1 + 0.09); 

    osc.connect(gain); 

    gain.connect(audioCtx.
