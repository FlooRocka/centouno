      };

 

      return pc;

    function listenToRoom() {

      roomRef.on('value', snapshot => {

        const data = snapshot.val();

        if (!data || data.status === 'ended') {

          returnToHomeView();

          return;

        }

 

        if (data.lastPlayerLeft && data.lastPlayerLeft.timestamp > lastSeenLeaveTimestamp) {

          lastSeenLeaveTimestamp = data.lastPlayerLeft.timestamp;

          if (data.lastPlayerLeft.id !== myId) {

            showLeaveToast(`⚠️ ${data.lastPlayerLeft.name} ha lasciato la stanza`);

          }

        }

 

        if (data.lastUnoCall && data.lastUnoCall.timestamp > lastSeenUnoTimestamp) {

          lastSeenUnoTimestamp = data.lastUnoCall.timestamp;

          soundUnoBroadcast();

        }

 

        if (data.lastRoundVictory && data.lastRoundVictory.timestamp > lastSeenVictoryTimestamp) {

          lastSeenVictoryTimestamp = data.lastRoundVictory.timestamp;

          soundVictoryFanfare();

        }

 

        if (data.voiceCallSignal && data.voiceCallSignal.senderId !== myId && data.voiceCallSignal.timestamp > lastSeenVoiceCallTimestamp) {

          lastSeenVoiceCallTimestamp = data.voiceCallSignal.timestamp;

          if (data.voiceCallSignal.type === 'start' && !isInVoiceCall) {

            document.getElementById('voice-invite-msg').textContent = `${data.voiceCallSignal.senderName} ha avviato la chat vocale. Vuoi unirti?`;

            document.getElementById('modal-voice-invite').classList.remove('hidden');

          }

        }

 

        if (data.players) {

          data.players = Array.isArray(data.players) ? data.players : Object.values(data.players);

          data.players.forEach(p => {

            if (!p.hand) p.hand = [];

            else if (!Array.isArray(p.hand)) p.hand = Object.values(p.hand);

          });

        }

        if (data.deck && !Array.isArray(data.deck)) data.deck = Object.values(data.deck);

        if (data.discardPile && !Array.isArray(data.discardPile)) data.discardPile = Object.values(data.discardPile);

 

        currentGameState = data;

        isHost = (data.hostId === myId);

 

        document.getElementById('room-id-badge').textContent = currentRoomCode;

 

        if (data.status === 'lobby') {

          showRoomModal();

          updateLobbyUI(data.players || []);

        } else if (data.status === 'playing') {

          document.getElementById('modal-home').classList.add('hidden');

          document.getElementById('modal-room').classList.add('hidden');

          document.getElementById('info-modal').classList.add('hidden');

          updateGameUI();

        } else if (data.status === 'round_over') {

          clearUnoTimer();

          showRoundOverModal(data.roundSummary, data.isGameEnd);

        }

      });

    }

    function isPlayable(card) {

      if (!currentGameState || !currentGameState.discardPile) return false;

      const top = currentGameState.discardPile[currentGameState.discardPile.length - 1];

 

      if (currentGameState.pendingDrawCount > 0) {

        return card.value === '2';

      }

 

      if (currentGameState.aceComboActive) {

        return card.value === 'A' || card.value === '9' || card.suit === currentGameState.activeSuit;

      }

 

      if (card.value === '9') return true;

 

      return card.suit === currentGameState.activeSuit || card.value === top.value;

    }

    function drawCardWithShuffle(deck, discard) {

      if (deck.length === 0) {

        if (discard.length > 1) {

          soundShuffleCards();

          const topVisible = discard[discard.length - 1];

          const cardsToShuffle = discard.slice(0, discard.length - 1);

          deck = cardsToShuffle.sort(() => Math.random() - 0.5);

          discard = [topVisible];

        }

      }

      const drawnCard = deck.length > 0 ? deck.pop() : null;

      return { card: drawnCard, deck: deck, discard: discard };

    }

          const endAngle = 370;

          angleDeg = startAngle + (i * (endAngle - startAngle) / (totalOthers - 1));

        const angleRad = (angleDeg * Math.PI) / 180;

        const rx = 44;

        const ry = 42;

        const posX = 50 + rx * Math.cos(angleRad);

        const posY = 50 + ry * Math.sin(angleRad);

 

        const isPlayerTurn = p.originalIndex === currentTurnIdx;

 

        const node = document.createElement('div');

        node.className = `avatar-node ${isPlayerTurn ? 'active-turn' : ''} ${p.eliminated ? 'eliminated' : ''}`;

        node.style.top = `${posY}%`;

            ${p.name.charAt(0).toUpperCase()}

            <div class="avatar-badge-cards">🂠 ${p.hand ? p.hand.length : 0}</div>

      const top = currentGameState.discardPile[currentGameState.discardPile.length - 1];

      renderCard(top, topCardEl);

 

      const currentDiscardCount = currentGameState.discardPile.length;

      if (currentDiscardCount !== previousDiscardCount) {

        topCardEl.classList.remove('discard-slam');

        void topCardEl.offsetWidth;

        topCardEl.classList.add('discard-slam');

        previousDiscardCount = currentDiscardCount;

      }

 

      const suitPill = document.getElementById('suit-pill-box');

      const suitDisplay = document.getElementById('active-suit-display');

      const activeSuit = currentGameState.activeSuit || '♥️';

      suitDisplay.textContent = activeSuit;

      

      const isRedSuit = (activeSuit === '♥️' || activeSuit === '♦️');

      suitPill.className = `active-suit-pill ${isRedSuit ? 'red' : 'black'}`;

      suitDisplay.style.color = isRedSuit ? '#f87171' : '#38bdf8';

 

      const me = players.find(p => p.id === myId);

      if (!me) return;

 

      const myHandEl = document.getElementById('my-hand');

      myHandEl.innerHTML = '';

      const myHand = me.hand || [];

      document.getElementById('my-cards-count').textContent = myHand.length;

 

      if (myHand.length > lastHandLength && currentGameState.discardPile.length > 1) {

        const prevTop = currentGameState.discardPile[currentGameState.discardPile.length - 2];

        if (prevTop && prevTop.value === '2') {

          soundTeasingPenalty();

        }

      }

      lastHandLength = myHand.length;

 

      if (myHand.length !== 1) {

        clearUnoTimer();

        hasCalledUnoSuccessfully = false;

        resetUnoBtnUI();

      }

 

      const n = myHand.length;

      const maxAngle = Math.min(36, n * 8);

      const angleStep = n > 1 ? maxAngle / (n - 1) : 0;

      const startAngle = -(maxAngle / 2);

      const cardSpacing = n > 4 ? Math.max(22, 180 / n) : 34;

 

      myHand.forEach((card, i) => {

        const cDiv = document.createElement('div');

        cDiv.className = `card fan-card ${card.red ? 'red' : ''}`;

        cDiv.innerHTML = `

          <div class="card-top">${card.value}<br>${card.suit}</div>

          <div class="card-mid">${card.suit}</div>

          <div class="card-bot">${card.value}</div>

        `;

 

        const rot = n > 1 ? (startAngle + (i * angleStep)) : 0;

        const xOffset = (i - (n - 1) / 2) * cardSpacing;

        const yOffset = Math.abs(rot) * 0.45;

        const baseTransform = `translateX(${xOffset}px) translateY(${yOffset}px) rotate(${rot}deg)`;

 

        cDiv.style.setProperty('--base-transform', baseTransform);

        cDiv.style.transform = baseTransform;

        cDiv.style.zIndex = i + 1;

 

        cDiv.onclick = (e) => onCardClick(i, cDiv);

        myHandEl.appendChild(cDiv);

      });

 

      const btnDraw = document.getElementById('btn-draw');

      btnDraw.disabled = !isMyTurn || me.eliminated;

      if (currentGameState.drawnThisTurn) {

        btnDraw.textContent = "Passa Turno";

        btnDraw.onclick = actionPass;

      } else {

        if (currentGameState.pendingDrawCount > 0) {

          btnDraw.textContent = `Pesca +${currentGameState.pendingDrawCount}`;

        } else {

          btnDraw.textContent = "Pesca carta";

        }

        btnDraw.onclick = actionDraw;

      }

    }

 

    function onCardClick(idx, cardElement) {

      const activePlayer = currentGameState.players[currentGameState.currentTurn];

      if (!activePlayer || activePlayer.id !== myId) return;

 

      const me = currentGameState.players.find(p => p.id === myId);

      const card = me.hand[idx];

 

      if (!isPlayable(card)) {

        return;

      }

 

      if (me.hand.length === 1 && !hasCalledUnoSuccessfully) {

        soundTeasingPenalty();

        alert("⚠️ PENALITÀ! Hai cercato di chiudere senza chiamare UNO: peschi 2 carte!");

        clearUnoTimer();

        applyUnoPenalty();

        return;

      }

 

      if (cardElement) {

        cardElement.classList.add('card-fly-out');

      }

 

      setTimeout(() => {

        if (card.value === '9' && currentGameState.aceComboActive) {

          executePlayCard(idx, currentGameState.activeSuit);

        } else if (card.value === '9') {

          pending9CardIndex = idx;

          document.getElementById('suit-modal').classList.remove('hidden');

        } else {

          executePlayCard(idx, null);

        }

      }, 160);

    }

 

    function chooseSuit(suit) {

      document.getElementById('suit-modal').classList.add('hidden');

      if (pending9CardIndex !== -1) {

        executePlayCard(pending9CardIndex, suit);

        pending9CardIndex = -1;

      }

    }

 

    function executePlayCard(cardIdx, chosenSuit) {

      soundPlayCard();

      let deck = currentGameState.deck ? [...currentGameState.deck] : [];

      let discard = [...(currentGameState.discardPile || [])];

      const players = currentGameState.players.map(p => ({

        ...p,

        hand: p.hand ? [...p.hand] : []

      }));

      const me = players.find(p => p.id === myId);

      const card = me.hand.splice(cardIdx, 1)[0];

 

      discard.push(card);

 

      let newSuit = (card.value === '9' && chosenSuit) ? chosenSuit : card.suit;

      if (card.value === '9' && currentGameState.aceComboActive) {

        newSuit = currentGameState.activeSuit;

      }

 

      let newPendingDraw = currentGameState.pendingDrawCount || 0;

      let newDirection = currentGameState.direction || 1;

      let nextTurn = currentGameState.currentTurn;

      let isAceCombo = false;

      let hasDrawn = false;

 

      if (card.value === '2') {

        newPendingDraw += 2;

      }

      if (card.value === '8') {

        newDirection *= -1;

      }

 

      if (me.hand.length === 1 && !hasCalledUnoSuccessfully) {

        startUnoTimer();

      }

 

      if (card.value === 'A') {

        isAceCombo = true;

        nextTurn = currentGameState.currentTurn;

 

        if (me.hand.length === 0) {

          const res = drawCardWithShuffle(deck, discard);

          deck = res.deck;

          discard = res.discard;

          if (res.card) {

            me.hand.push(res.card);

          }

          hasDrawn = true;

          if (me.hand.length === 1 && !hasCalledUnoSuccessfully) {

            startUnoTimer();

          }

        }

      } else {

        isAceCombo = false;

 

        if (me.hand.length === 0) {

          clearUnoTimer();

 

          let victim = null;

          if (card.value === '2' && newPendingDraw > 0) {

            const myIndex = players.findIndex(p => p.id === myId);

            let nextIndex = myIndex;

            do {

              nextIndex = (nextIndex + newDirection + players.length) % players.length;

            } while (players[nextIndex].eliminated && nextIndex !== myIndex);

 

            if (nextIndex !== myIndex) {

              victim = players[nextIndex];

              for (let i = 0; i < newPendingDraw; i++) {

                const res = drawCardWithShuffle(deck, discard);

                deck = res.deck;

                discard = res.discard;

                if (res.card) {

                  victim.hand.push(res.card);

                }

              }

            }

          }

 

          handleEndRound(me, players, victim, newPendingDraw);

          return;

        }

 

        const activePlayersCount = players.filter(p => !p.eliminated).length;

        if (card.value === '8' && activePlayersCount === 2) {

          // Con 2 giocatori l'8 fa rigiocare

        } else {

          do {

            nextTurn = (nextTurn + newDirection + players.length) % players.length;

          } while (players[nextTurn].eliminated);

        }

      }

 

      roomRef.update({

        players: players,

        deck: deck,

        discardPile: discard,

        activeSuit: newSuit,

        pendingDrawCount: newPendingDraw,

        direction: newDirection,

        aceComboActive: isAceCombo,

        drawnThisTurn: hasDrawn,

        currentTurn: nextTurn

      });

    }

 

    function actionDraw() {

      const activePlayer = currentGameState.players[currentGameState.currentTurn];

      if (!activePlayer || activePlayer.id !== myId) return;

 

      clearUnoTimer();

      soundDrawCard();

 

      const deckEl = document.getElementById('deck');

      if (deckEl) {

        deckEl.classList.remove('deck-draw-pop');

        void deckEl.offsetWidth;

        deckEl.classList.add('deck-draw-pop');

      }

 

      let deck = currentGameState.deck ? [...currentGameState.deck] : [];

      let discard = currentGameState.discardPile ? [...currentGameState.discardPile] : [];

      const players = currentGameState.players.map(p => ({

        ...p,

        hand: p.hand ? [...p.hand] : []

      }));

      const me = players.find(p => p.id === myId);

 

      const countToDraw = currentGameState.pendingDrawCount > 0 ? currentGameState.pendingDrawCount : 1;

 

      for (let i = 0; i < countToDraw; i++) {

        const res = drawCardWithShuffle(deck, discard);

        deck = res.deck;

        discard = res.discard;

        if (res.card) {

          me.hand.push(res.card);

        }

      }

 

      roomRef.update({

        deck: deck,

        discardPile: discard,

        players: players,

        pendingDrawCount: 0,

        drawnThisTurn: true

      });

    }

 

    function actionPass() {

      clearUnoTimer();

      soundTurnClose();

 

      const players = currentGameState.players;

      let nextTurn = currentGameState.currentTurn;

      do {

        nextTurn = (nextTurn + currentGameState.direction + players.length) % players.length;

      } while (players[nextTurn].eliminated);

 

      roomRef.update({ 

        currentTurn: nextTurn,

        aceComboActive: false,

        drawnThisTurn: false 

      });

    }

 

    function handleEndRound(winner, players, victim, penaltyCardsCount) {

      soundShuffleCards();

      let summary = `${winner.name} vince la mano!\n`;

 

      if (victim && penaltyCardsCount > 0) {

        summary += `💥 Chiusura col 2! ${victim.name} ha pescato ${penaltyCardsCount} carte prima del conteggio!\n`;

      }

 

      summary += `\nPunteggi aggiunti:\n`;

      const winnerIdx = players.findIndex(p => p.id === winner.id);

 

      players.forEach(p => {

        if (!p.eliminated) {

          const handPts = (p.hand || []).reduce((sum, c) => sum + getPoints(c.value), 0);

          p.score = (p.score || 0) + handPts;

          summary += `• ${p.name}: +${handPts} pt (Tot: ${p.score})\n`;

 

          if (p.score >= 101) {

            p.eliminated = true;

            summary += `  ❌ ${p.name} eliminato!\n`;

          }

        }

      });

 

      const remaining = players.filter(p => !p.eliminated);

      const isGameEnd = remaining.length <= 1;

      if (isGameEnd) {

        summary += `\n🏆 VINCITORE DELLA PARTITA: ${remaining[0] ? remaining[0].name : 'Nessuno'}!`;

      }

 

      roomRef.update({

        status: 'round_over',

        players: players,

        roundSummary: summary,

        isGameEnd: isGameEnd,

        lastWinnerIndex: winnerIdx,

        lastRoundVictory: {

          timestamp: Date.now()

        }

      });

    }

    function nextRoundAction() {

      if (!isHost) return;

      if (currentGameState.isGameEnd) {

        const players = currentGameState.players.map(p => ({

          ...p, score: 0, eliminated: false, hand: []

        }));

        roomRef.update({ status: 'lobby', players: players });

      } else {

        soundShuffleCards();

 

        const playersList = currentGameState.players;

        const deck = createDeckForPlayerCount(playersList.length);

 

        playersList.forEach(p => {

          p.hand = [];

          if (!p.eliminated) {

            for (let i = 0; i < 5; i++) {

              p.hand.push(deck.pop());

            }

          }

        });

 

        let firstCard = deck.pop();

        while (['A', '2', '8', '9'].includes(firstCard.value)) {

          deck.unshift(firstCard);

          firstCard = deck.pop();

        }

 

        const winnerIndex = typeof currentGameState.lastWinnerIndex === 'number' 

          ? currentGameState.lastWinnerIndex 

          : 0;

 

        let nextStartingTurn = winnerIndex;

        do {

          nextStartingTurn = (nextStartingTurn + 1) % playersList.length;

        } while (playersList[nextStartingTurn].eliminated);

 

        hasCalledUnoSuccessfully = false;

        resetUnoBtnUI();

 

        roomRef.update({

          status: 'playing',

          deck: deck,

          discardPile: [firstCard],

          activeSuit: firstCard.suit,

          currentTurn: nextStartingTurn,

          direction: 1,

          pendingDrawCount: 0,

          aceComboActive: false,

          drawnThisTurn: false,

          players: playersList,

          cardsPerPlayer: 5

        });

      }

    }
